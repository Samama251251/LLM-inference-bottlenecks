# Phase 1 dependencies, Environment A (the HuggingFace baseline).
#
# Two environments on purpose. These pins cover the baseline only. vLLM brings
# its own torch matched to its CUDA build, so installing it here would replace
# the torch the baseline was measured on. It gets an isolated venv; see README.
#
# Do NOT pin torch here. Use whatever the box's image ships, matched to its
# CUDA, and let the CSV record which version actually ran. The versions that
# have been measured on are in the README table.

# --- HuggingFace baseline ---
# transformers 4.44.2 used the legacy per-layer rotary embedding that registers
# cos_cached/sin_cached buffers sized to max_position_embeddings (131072 for
# Qwen2.5), duplicated across all 28 layers. That is a fixed 1.79 GB of VRAM
# (28 layers x 2 tables x 131072 x 128 x 2 bytes) that has nothing to do with
# the weights and never gets used past the current sequence length. 4.46+ moved
# rotary to a single model-level module that computes cos/sin on the fly, so the
# buffers vanish and resident weights drop to the clean ~2944 MiB fp16 floor.
# test/probe_weights_vram.py is the diagnostic that pinned this down.
transformers==4.46.3
accelerate==0.34.2
# Qwen2.5 uses a fast tokenizer, so no sentencepiece is needed. tokenizers must
# be on 0.20.x to satisfy transformers 4.46.
tokenizers==0.20.3
safetensors==0.4.4

# --- plotting and analysis ---
# The plotting scripts import only matplotlib and the stdlib, deliberately, so
# figures regenerate from committed CSVs on a laptop with no GPU and no torch.
matplotlib==3.9.2

# --- NVML, for reading device memory on the vLLM path ---
# vLLM's v1 engine runs the model in a child process, so torch.cuda counters
# read 0 from the parent. NVML sees the whole device. Falls back to nvidia-smi.
nvidia-ml-py==12.560.30

# --- Environment B: vLLM, installed separately ---

#   vllm==0.24.0, with the torch it pulls in (2.11.0, cu129 or cu130)
#
#   python -m venv vllm-env && source vllm-env/bin/activate
#   pip install uv
#   uv pip install vllm==0.24.0 --torch-backend=auto
#
# --torch-backend=auto matches the box's CUDA. All cards measured so far are
# compute capability 8.0+, so the optimized attention backends are available
# and no fallback is needed.
